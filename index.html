// ======================================================== // LOGIKA TAMPILAN PGM BERDASARKAN 3 STATUS (OFF / STANDBY / ON-AIR) // ======================================================== function updateMonitors() { const isRoomLive = roomLiveStatesMap.get(currentRoomToken) === true || currentRoomToken === 'SKASA'; // Asumsi status OFF dapat dideteksi jika meta.status === 'ended' atau tidak ada data aktif const roomMeta = activeRoomsMap.get(currentRoomToken) || {}; const isRoomOff = roomMeta.status === 'ended'; const pgmVideo = document.getElementById('pgmCamVideo'); const pgmFallback = document.getElementById('pgmFallbackBox'); const pgmVtWrap = document.getElementById('pgmVtWrapper'); const standbyOverlay = document.getElementById('pgmStandbyCardOverlay'); // Elemen Kartu Nama Room const lblActiveSrc = document.getElementById('lblActivePgmSource'); const screenLabel = document.getElementById('pgmScreenLabel'); // 1. KONDISI: STATUS OFF (Tampil Looping Bumper Penuh) if (isRoomOff) { if (lblActiveSrc) lblActiveSrc.innerText = 'OFF-AIR (BUMPER IDLE)'; if (standbyOverlay) standbyOverlay.classList.add('hidden'); if (pgmVtWrap) { pgmVtWrap.classList.remove('opacity-0', 'pointer-events-none'); pgmVtWrap.classList.add('opacity-100'); pgmVtWrap.style.zIndex = '15'; } // Putar bumper default secara loop if (!latestVtState.youtubeId) latestVtState.youtubeId = DEFAULT_BUMPER_YT_ID; latestVtState.status = 'playing'; syncVtDisplay(true); setSyncIndicator(false, 'Off-Air Bumper'); return; } // 2. KONDISI: STATUS STANDBY (Looping Bumper + Kartu Nama Room Transparan) if (!isRoomLive) { if (lblActiveSrc) lblActiveSrc.innerText = 'STANDBY (GLADI BERSIH)'; if (screenLabel) screenLabel.innerText = currentRoomToken; // Perbarui teks kartu nama room & program aktif pada overlay transparan const cardTokenEl = document.getElementById('standbyOverlayToken'); const cardProgramEl = document.getElementById('standbyOverlayProgram'); if (cardTokenEl) cardTokenEl.innerText = currentRoomToken; if (cardProgramEl) cardProgramEl.innerText = roomMeta.programName || "Gladi Bersih Studio"; // Tampilkan kartu identitas dengan efek opacity if (standbyOverlay) standbyOverlay.classList.remove('hidden'); // PGM tetap memutar Bumper/VT di latar belakang if (pgmVideo) { pgmVideo.classList.remove('opacity-100'); pgmVideo.classList.add('opacity-0', 'pointer-events-none'); } if (pgmFallback) pgmFallback.classList.add('hidden'); if (pgmVtWrap) { pgmVtWrap.classList.remove('opacity-0', 'pointer-events-none'); pgmVtWrap.classList.add('opacity-100'); pgmVtWrap.style.zIndex = '10'; // Di bawah kartu overlay } if (!latestVtState.youtubeId) latestVtState.youtubeId = DEFAULT_BUMPER_YT_ID; latestVtState.status = 'playing'; syncVtDisplay(true); setSyncIndicator(false, 'Standby (Gladi Bersih)'); return; } // 3. KONDISI: STATUS ON-AIR (Tampil PGM Utama / Switcher Aktif) if (standbyOverlay) standbyOverlay.classList.add('hidden'); const activeSrc = liveState.programSource || 'cam_1'; const pgmLabel = activeSrc.toUpperCase().replace('_', ' '); if (lblActiveSrc) lblActiveSrc.innerText = pgmLabel; if (screenLabel) screenLabel.innerText = pgmLabel; syncVtDisplay(false); if (activeSrc === 'vt') { if (pgmVideo) { pgmVideo.classList.remove('opacity-100'); pgmVideo.classList.add('opacity-0', 'pointer-events-none'); } if (pgmFallback) pgmFallback.classList.add('hidden'); if (pgmVtWrap) { pgmVtWrap.classList.remove('opacity-0', 'pointer-events-none'); pgmVtWrap.classList.add('opacity-100'); pgmVtWrap.style.zIndex = '15'; } if (isAudioUnmuted) { sendCommandToYtIframe('unMute'); sendCommandToYtIframe('setVolume', [100]); } setSyncIndicator(true, 'VT Playout On-Air'); } else { if (pgmVtWrap) { pgmVtWrap.classList.remove('opacity-100'); pgmVtWrap.classList.add('opacity-0', 'pointer-events-none'); pgmVtWrap.style.zIndex = '0'; } sendCommandToYtIframe('mute'); const stream = getCameraStream(activeSrc); if (stream && pgmVideo) { if (pgmVideo.srcObject !== stream) pgmVideo.srcObject = stream; pgmVideo.classList.remove('opacity-0', 'pointer-events-none'); pgmVideo.classList.add('opacity-100'); pgmVideo.style.zIndex = '15'; if (pgmFallback) pgmFallback.classList.add('hidden'); pgmVideo.play().catch(() => {}); setSyncIndicator(true, `Kamera ${pgmLabel} Aktif`); } else { if (pgmVideo) { pgmVideo.classList.remove('opacity-100'); pgmVideo.classList.add('opacity-0', 'pointer-events-none'); } if (pgmFallback) pgmFallback.classList.remove('hidden'); setSyncIndicator(false, `Menyambungkan ${pgmLabel}`); } } }
