// ======================================================== // DAFTAR SALURAN AKTIF & STATUS REAL-TIME ON AIR/STANDBY --> // ======================================================== const activeRoomsMap = new Map(); const roomLiveStatesMap = new Map(); function listenChannelsDirectory() { // Inisialisasi default awal activeRoomsMap.set('SKASA', { programName: "SKASA TV Live Broadcast", episode: "Siaran Utama Sekolah SMKN 1 Pacitan", status: "live" }); roomLiveStatesMap.set('SKASA', true); // SKASA sebagai ruang utama default renderChannelsGrid(); const activeDb = getDatabase(); if (!activeDb) return; // Pantau data ruangan secara real-time dari Firebase activeDb.ref('rooms').on('value', snapshot => { if (!snapshot.exists()) return; const rooms = snapshot.val() || {}; Object.keys(rooms).forEach(token => { const roomData = rooms[token]; if (roomData && roomData.meta) { activeRoomsMap.set(token, roomData.meta); } // [PERBAIKAN] Validasi ketat status On-Air / Standby aktual let isLive = false; if (roomData.live_state) { // Hanya aktif ON-AIR jika isLive bernilai true secara eksplisit isLive = roomData.live_state.isLive === true; } else if (roomData.status === 'LIVE' || roomData.isLive === true) { isLive = true; } roomLiveStatesMap.set(token, isLive); }); renderChannelsGrid(); }); } function renderChannelsGrid() { const grid = document.getElementById('channelRoomsGrid'); if (!grid) return; grid.innerHTML = ''; let keys = Array.from(activeRoomsMap.keys()); if (!keys.includes('SKASA')) keys.unshift('SKASA'); else { const idx = keys.indexOf('SKASA'); if (idx > 0) { keys.splice(idx, 1); keys.unshift('SKASA'); } } const countEl = document.getElementById('lblChannelCount'); if (countEl) countEl.innerText = `${keys.length} Saluran Siaran`; keys.forEach(token => { const meta = activeRoomsMap.get(token) || {}; const isSkasa = token === 'SKASA'; // [PERBAIKAN] Evaluasi akurat sesuai status real-time Firebase const isLive = roomLiveStatesMap.get(token) === true; const isSelected = token === currentRoomToken; const card = document.createElement('div'); card.onclick = () => { if (token === currentRoomToken) return; const newUrl = `${window.location.pathname}?room=${token}`; window.history.pushState({ path: newUrl }, '', newUrl); connectChannel(token); renderChannelsGrid(); }; let cardStyle = "p-3.5 rounded-xl border transition cursor-pointer flex items-center justify-between "; if (isSelected) { cardStyle += "bg-red-950/60 border-red-500 shadow-lg shadow-red-950/50 ring-1 ring-red-500"; } else { cardStyle += "bg-slate-950 border-slate-800 hover:border-slate-700 hover:bg-slate-900"; } card.className = cardStyle; card.innerHTML = `
${isLive ? 'LIVE' : 'CH'}
${token} ${isSkasa ? 'UTAMA' : ''}
${meta.programName || "SKASA TV Broadcast"}
${isLive ? '● ON-AIR' : 'STANDBY'}
`; grid.appendChild(card); }); }
